// KALICI PROFİL VE OTURUM KONTROLÜ function checkUserAuth() { // Hem localStorage hem sessionStorage kontrol edilir (Asla sıfırlanmaz) const session = JSON.parse(localStorage.getItem('active_session') || sessionStorage.getItem('active_session') || 'null'); const navAuthBtn = document.getElementById('navAuthBtn'); const userAvatarBtn = document.getElementById('userAvatarBtn'); const navScoreChip = document.getElementById('navScoreChip'); const drawerProfileBox = document.getElementById('drawerProfileBox'); const drawerRegisterLink = document.getElementById('drawerRegisterLink'); const adminMenuItem = document.getElementById('adminScoreMenuItem'); if (!session) { if (navAuthBtn) navAuthBtn.style.display = 'inline-flex'; if (userAvatarBtn) userAvatarBtn.style.display = 'none'; if (navScoreChip) navScoreChip.style.display = 'none'; if (drawerProfileBox) drawerProfileBox.style.display = 'none'; if (adminMenuItem) adminMenuItem.style.display = 'none'; if (drawerRegisterLink) drawerRegisterLink.style.display = 'flex'; } else { // Oturumu iki tarafa da sabitle localStorage.setItem('active_session', JSON.stringify(session)); if (navAuthBtn) navAuthBtn.style.display = 'none'; if (userAvatarBtn) userAvatarBtn.style.display = 'flex'; if (drawerProfileBox) drawerProfileBox.style.display = 'flex'; if (drawerRegisterLink) drawerRegisterLink.style.display = 'none'; if (session.role === 'admin' || session.username === 'admin') { // YÖNETİCİ GİRİŞİ if (navScoreChip) navScoreChip.style.display = 'none'; if (adminMenuItem) adminMenuItem.style.display = 'block'; const nameEl = document.getElementById('drawerUserName'); const deptEl = document.getElementById('drawerUserDept'); const scoreCont = document.getElementById('drawerScoreContainer'); if (nameEl) nameEl.innerText = "Yönetici"; if (deptEl) deptEl.innerText = "Sistem Kontrol"; if (scoreCont) scoreCont.style.display = 'none'; } else { // NORMAL ÖĞRENCİ GİRİŞİ if (adminMenuItem) adminMenuItem.style.display = 'none'; const users = JSON.parse(localStorage.getItem('deu_members') || '[]'); const me = users.find(u => u.username === session.username); if (me) { if (navScoreChip) { navScoreChip.style.display = 'inline-flex'; const scoreVal = document.getElementById('navScoreValue'); if (scoreVal) scoreVal.innerText = `${me.points || me.score || 0} P`; } const nameEl = document.getElementById('drawerUserName'); const deptEl = document.getElementById('drawerUserDept'); const scoreCont = document.getElementById('drawerScoreContainer'); const ptsEl = document.getElementById('drawerUserPoints'); if (nameEl) nameEl.innerText = me.fullName || me.name; if (deptEl) deptEl.innerText = me.dept || 'Özel Eğitim'; if (scoreCont) scoreCont.style.display = 'inline-flex'; if (ptsEl) ptsEl.innerText = me.points || me.score || 0; if (me.avatar) { applyAvatarImage(me.avatar); } else { clearAvatarImage(); } } } } } window.userLogout = function() { localStorage.removeItem('active_session'); sessionStorage.removeItem('active_session'); window.location.reload(); }; checkUserAuth();
